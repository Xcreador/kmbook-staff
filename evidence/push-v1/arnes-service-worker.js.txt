const { chromium } = require('playwright');
const http = require('http'); const fs = require('fs'); const path = require('path');
const root = '/tmp/claude-0/staffsw';
const srv = http.createServer((q, r) => { let p = path.join(root, q.url.split('?')[0]); if (!fs.existsSync(p) || fs.statSync(p).isDirectory()) p = path.join(root, 'index.html'); r.writeHead(200, { 'content-type': p.endsWith('.js') ? 'text/javascript' : 'text/html' }); fs.createReadStream(p).pipe(r); }).listen(4518);
(async () => {
  const browser = await chromium.launch({ channel: 'chromium' });
  const ctx = await browser.newContext(); await ctx.grantPermissions(['notifications'], { origin: 'http://localhost:4518' });
  const page = await ctx.newPage(); await page.goto('http://localhost:4518/today');
  await page.evaluate(async () => { await navigator.serviceWorker.register('/sw.js', { scope: '/' }); await navigator.serviceWorker.ready; });
  const cdp = await ctx.newCDPSession(page); await cdp.send('ServiceWorker.enable');
  const regs = []; cdp.on('ServiceWorker.workerRegistrationUpdated', e => regs.push(...e.registrations));
  await new Promise(r => setTimeout(r, 500));
  const reg = regs.find(r => r.scopeURL.endsWith(':4518/'));
  const appt = '11111111-1111-4111-8111-111111111111';
  // Payload EXACTO del worker de Core (prepareNotification): title, body, url, tag, data{eventType,url}
  const payloads = [
    { title: 'Nueva reserva online', body: 'Hay una nueva reserva en la agenda.', url: `/app/studio/agenda?organization=o&appointment=${appt}`, tag: 'kmbook-appointment-booked_online-1', data: { eventType: 'appointment.booked_online', url: `/app/studio/agenda?organization=o&appointment=${appt}` } },
    { title: 'Tarea asignada', body: 'Tienes una tarea nueva.', url: '/app/studio/tasks?organization=o', tag: 'kmbook-task-assigned-2', data: { eventType: 'task.assigned' } },
    { title: 'Notificación de prueba', body: 'x', url: '/app/notifications/push?organization=o', tag: 'kmbook-push-test', data: { eventType: 'push.test' } },
  ];
  for (const p of payloads) await cdp.send('ServiceWorker.deliverPushMessage', { origin: 'http://localhost:4518', registrationId: reg.registrationId, data: JSON.stringify(p) });
  await new Promise(r => setTimeout(r, 1000));
  const shown = await page.evaluate(async () => (await (await navigator.serviceWorker.getRegistration('/')).getNotifications()).map(n => ({ title: n.title, tag: n.tag, url: n.data.url })));
  console.log(JSON.stringify(shown, null, 1));
  await browser.close(); srv.close();
})().catch(e => { console.error(e); process.exit(1); });
